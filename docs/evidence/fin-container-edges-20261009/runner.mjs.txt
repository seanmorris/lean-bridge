import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { execFileSync, spawn } from 'node:child_process';
import { closeSync, existsSync, mkdirSync, openSync, readFileSync, statfsSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';

// Prepared only: CPU3 must be explicitly assigned after the current Array producer is terminal.
const cwd = '/app/build/worktrees/fin-container-edges-producer-dc6b8b8';
const revision = 'dc6b8b8d615c6cf1073d697057cd012c6a1c4b3b';
const output = '/app/build/vo1454-fin-container-edges-dc6b8b8';
const floorMiB = 2048;
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const git = args => execFileSync('git', args, { cwd, encoding: 'utf8', maxBuffer: 16 * 1024 * 1024 }).trim();
const clean = () => {
  assert.equal(git(['rev-parse', 'HEAD']), revision);
  assert.equal(git(['status', '--porcelain', '--untracked-files=no']), '');
};
const freeMiB = () => { const disk = statfsSync(cwd); return Math.floor(disk.bavail * disk.bsize / 1024 ** 2); };
const tool = (command, args) => execFileSync(command, args, { encoding: 'utf8' }).trim().split('\n')[0];
clean(); assert.ok(freeMiB() >= floorMiB); mkdirSync(output);
const paths = [
  'tests/fin-container-edges.test.mjs', 'tests/helpers/fin-container-edges.mjs', 'tests/helpers/fin-container-edge-install.mjs',
  'tests/fixtures/fin-container-edges.lean', 'tests/fixtures/onboarding/native-fin-containers/FinContainers.lean',
  ...['c.c', 'cpp.cpp', 'python.py'].flatMap(name => ['tests/fixtures/fin-container-consumers/' + name, 'tests/fixtures/fin-container-edge-consumers/' + name]),
  'tests/helpers/fin-container-install.mjs', 'tests/helpers/copied-fixture-install.mjs', 'tests/helpers/package-set.mjs',
  'tests/helpers/lake-workspace.mjs', 'src/build/canonical-build.mjs', 'src/build/native-model.mjs', 'src/build/native-artifacts.mjs',
  'src/backends/c/gmp-projection.mjs', 'src/backends/c/gmp-values.mjs', 'src/backends/cpp/primitives.mjs', 'src/backends/cpp/copied-values.mjs',
  'src/backends/python/generate.mjs', 'src/backends/python/copied-conversions.mjs',
  'src/release/native-c-family.mjs', 'src/release/package-set-receipt.mjs'
];
const sources = Object.fromEntries(paths.map(path => {
  const current = readFileSync(join(cwd, path));
  assert.ok(current.equals(execFileSync('git', ['show', `${revision}:${path}`], { cwd, maxBuffer: 16 * 1024 * 1024 })), path);
  return [path, sha(current)];
}));
const expected = {
  c: { checks: 14114, consumerSha256: 'b9df0c88e8085e2ea605b4295811c966991e3efb6186f02a168f8224a751eb4d' },
  cpp: { checks: 14099, consumerSha256: '2243247b9aa590e001ee9b9290e6e00ca1914f2031a9bf2f9ce947b6bd53dead' },
  python: { checks: 14095, consumerSha256: '9e32423234dc8e8ab93451cba3858de7a55b9da23df980e8d076dea0f2ad1ba7' }
};
const selections = [
  { name: 'c-cpp', profiles: ['c', 'cpp'], report: 'edges-c-cpp.json' },
  { name: 'python311', profiles: ['python'], report: 'edges-python311.json', python: '/app/.toolchains/python311/bin/python3', version: '3.11.16' },
  { name: 'python312', profiles: ['python'], report: 'edges-python312.json', python: '/app/.toolchains/python312/bin/python3', version: '3.12.14' }
];
const lean = '/app/.toolchains/elan/toolchains/leanprover--lean4---v4.32.2';
const environmentFor = selection => ({ NO_COLOR: '1', LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR: '2.36', LEAN_BRIDGE_LEAN_PREFIX: lean,
  LEAN_NUM_THREADS: '1', OMP_NUM_THREADS: '1', MAKEFLAGS: '-j1',
  LEAN_BRIDGE_FIN_CONTAINER_EDGE_PROFILES: selection.profiles.join(','), LEAN_BRIDGE_FIN_CONTAINER_EDGE_REPORT: join(output, selection.report),
  ...(selection.python ? { LEAN_BRIDGE_PYTHON: selection.python } : {}) });
const base = Object.fromEntries(Object.entries(process.env).filter(([key]) => !key.startsWith('LEAN_BRIDGE_') && key !== 'FORCE_COLOR'));
const pattern = '^installed source-free native packages execute zero-bound and nested-position Fin edge cases$';
const args = ['-c', '3', process.execPath, '--test', '--test-concurrency=1', '--test-reporter=tap', `--test-name-pattern=${pattern}`, 'tests/fin-container-edges.test.mjs'];
const queue = { revision, tree: git(['rev-parse', 'HEAD^{tree}']), cwd, queuedAt: new Date().toISOString(), sources, expected,
  runnerSha256: sha(readFileSync(import.meta.filename)), command: ['/usr/bin/taskset', ...args],
  selections: selections.map(selection => ({ ...selection, environment: environmentFor(selection) })),
  node: process.version, lean: tool(join(lean, 'bin/lean'), ['--version']), glibc: tool('/usr/bin/getconf', ['GNU_LIBC_VERSION']),
  cc: tool('/usr/bin/cc', ['--version']), cxx: tool('/usr/bin/c++', ['--version']), readelf: tool('/usr/bin/readelf', ['--version']),
  pythons: Object.fromEntries(selections.filter(s => s.python).map(s => [s.python, tool(s.python, ['--version'])])),
  freeMiB: freeMiB(), stopFloorMiB: floorMiB,
  scope: 'Local ordinary-source installed C/C++/Python Fin zero-bound and nested-position behavior, source-free offline two-root packages and complete installed-tree relocation. glibc2.36 only. Not hosted, release-floor, other-host or expanded dispatch acceptance.' };
writeFileSync(join(output, 'queue.json'), JSON.stringify(queue, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify({ output, revision, queuedAt: queue.queuedAt, freeMiB: queue.freeMiB }));
for(const selection of selections) {
  clean(); assert.ok(freeMiB() >= floorMiB, `Need at least 2 GiB before ${selection.name}`);
  assert.equal(existsSync(join(output, selection.report)), false);
  const directory = join(output, selection.name); mkdirSync(directory);
  const log = openSync(join(directory, 'run.tap'), 'wx'), startedAt = new Date().toISOString();
  let minimumFreeMiB = freeMiB(), stoppedForDisk = false, escalation;
  const child = spawn('/usr/bin/taskset', args, { cwd, env: { ...base, ...environmentFor(selection) }, detached: true, stdio: ['ignore', log, log] });
  writeFileSync(join(directory, 'start.json'), JSON.stringify({ startedAt, pid: child.pid, pgid: child.pid }, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ selection: selection.name, startedAt, pid: child.pid }));
  const monitor = setInterval(() => {
    minimumFreeMiB = Math.min(minimumFreeMiB, freeMiB());
    if(minimumFreeMiB < floorMiB && !stoppedForDisk) {
      stoppedForDisk = true; process.kill(-child.pid, 'SIGTERM');
      escalation = setTimeout(() => { try { process.kill(-child.pid, 'SIGKILL'); } catch {} }, 10000);
    }
  }, 5000);
  const result = await new Promise(resolve => {
    child.once('error', error => resolve({ code: null, signal: null, error: error.message }));
    child.once('close', (code, signal) => resolve({ code, signal }));
  });
  clearInterval(monitor); clearTimeout(escalation); closeSync(log);
  minimumFreeMiB = Math.min(minimumFreeMiB, freeMiB());
  const end = { ...result, stoppedForDisk, minimumFreeMiB, endedAt: new Date().toISOString(), tapSha256: sha(readFileSync(join(directory, 'run.tap'))) };
  writeFileSync(join(directory, 'end.json'), JSON.stringify(end, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ selection: selection.name, ...end }));
  clean(); assert.equal(stoppedForDisk, false); assert.equal(result.code, 0, selection.name);
  const tap = readFileSync(join(directory, 'run.tap'), 'utf8').split('\n');
  for(const line of ['# tests 1', '# pass 1', '# fail 0', '# skipped 0', '# cancelled 0', '# todo 0']) assert.ok(tap.includes(line), `${selection.name}: ${line}`);
  const bytes = readFileSync(join(output, selection.report)), report = JSON.parse(bytes);
  assert.deepEqual([report.schemaVersion, report.profiles, report.reproducible, report.authorRoots], [1, selection.profiles, true, 2]);
  assert.deepEqual(report.reports.map(item => item.profile), selection.profiles);
  for(const item of report.reports) {
    assert.equal(item.path, 'ordinary-source'); assert.equal(item.checks, expected[item.profile].checks);
    assert.equal(item.consumerSha256, expected[item.profile].consumerSha256);
    for(const flag of ['offlineInstall', 'compilerFreePath', 'sourceRemovedBeforeInstallation', 'relocatedInstallation', 'repeatExecution', 'installedFilesUnchanged']) assert.equal(item[flag], true, flag);
    for(const field of ['fixtureSha256', 'bindingIrSha256', 'sourceTreeSha256', 'modelSha256', 'receiptSha256', 'installedFilesSha256', 'installedReceiptSha256']) assert.match(item[field], /^[a-f0-9]{64}$/u, field);
    assert.equal(item.dispatch.observed, false);
    if(item.profile === 'python') assert.equal(item.python, selection.version);
    else {
      assert.match(item.executableSha256, /^[a-f0-9]{64}$/u);
      assert.equal(item.runtimeSearchPath, `$ORIGIN/fincontainers-1.0.0-${item.profile}/lib`);
    }
    for(const pkg of item.packages) for(const artifact of pkg.artifacts) assert.equal(report.archives[artifact.path], artifact.sha256);
  }
  console.log(JSON.stringify({ selection: selection.name, report: selection.report, sha256: sha(bytes), checks: report.reports.map(item => [item.profile, item.checks]) }));
}
