import assert from 'node:assert/strict';
import { execFileSync, spawn } from 'node:child_process';
import { cp, mkdir, open, readFile, rm, statfs, writeFile } from 'node:fs/promises';
import { join, resolve } from 'node:path';
import { canonicalJson, sha256 } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/src/capsule/node.mjs';
import { buildCanonicalProject } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/src/build/canonical-build.mjs';
import { verifyPackageSetReceipt } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/src/release/package-set-receipt.mjs';
import { copyPackageSetHandoff } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/package-set.mjs';
import { finContainerEnvironment, finContainerTargets } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/fin-container-install.mjs';
import { finContainerEdgeSource, finContainerEdgeRefinements, finContainerEdgeConsumer } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/fin-container-edges.mjs';
import { installCopiedConsumer } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/copied-fixture-install.mjs';
import { installFinContainerEdgeJvm } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/fin-container-edge-jvm-closure.mjs';
import { verifyFinContainerEdgeArchiveClosure } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/fin-container-edge-closure.mjs';
import { repeatFinContainerEdges } from '/app/build/worktrees/fin-container-edge-file-closure-vo1454/tests/helpers/fin-container-edge-install.mjs';

const revision = process.argv[2];
assert.match(revision ?? '', /^[a-f0-9]{40}$/);
const git = args => execFileSync('git', args, { encoding: 'utf8', maxBuffer: 32 * 1024 * 1024 });
const clean = () => {
  assert.equal(git(['rev-parse', 'HEAD']).trim(), revision);
  assert.equal(git(['diff', '--name-only', 'HEAD']).trim(), '');
};
const free = async () => { const s = await statfs('/app'); return Math.floor(s.bavail * s.bsize / 1024 / 1024); };
const root = resolve('/app/build', `vo1454-actual-edge-jvm-${revision.slice(0, 7)}`);
const save = (name, value) => writeFile(join(root, name), JSON.stringify(value, null, 2) + '\n', { flag: 'wx' });
const scope = 'Local diagnostic preparation of the exact original-plus-zero/nested-edge Lean source, canonical Maven JAR, full Java edge Consumer and eight-column public/adapter observer. One author root, glibc floor 2.36. Not two-root or hosted acceptance and not a debugger fix.';
clean();
if (process.argv[3] !== '--worker') {
  assert.ok(await free() >= 2048, 'Need 2 GiB free before starting');
  await mkdir(root);
  const log = await open(join(root, 'run.log'), 'wx');
  const child = spawn('/usr/bin/taskset', ['-c', '3', process.execPath, import.meta.filename, revision, '--worker'], {
    cwd: process.cwd(), detached: true, stdio: ['ignore', log.fd, log.fd],
    env: { ...process.env, NO_COLOR: '1', LEAN_NUM_THREADS: '1', OMP_NUM_THREADS: '1', MAKEFLAGS: '-j1', LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR: '2.36' }
  });
  await save('start.json', { revision, tree: git(['rev-parse', 'HEAD^{tree}']).trim(), pid: child.pid,
    startedAt: new Date().toISOString(), freeMiB: await free(), stopFloorMiB: 768, scope,
    runnerSha256: sha256(await readFile(import.meta.filename)) });
  console.log(JSON.stringify({ root, pid: child.pid }));
  let minimumFreeMiB = await free(), stoppedForDisk = false, escalation;
  const signal = value => { try { process.kill(-child.pid, value); } catch (e) { if (e.code !== 'ESRCH') throw e; } };
  const monitor = setInterval(async () => {
    minimumFreeMiB = Math.min(minimumFreeMiB, await free());
    if (!stoppedForDisk && minimumFreeMiB < 768) {
      stoppedForDisk = true; signal('SIGTERM'); escalation = setTimeout(() => signal('SIGKILL'), 10000);
    }
  }, 3000);
  const result = await new Promise((resolve, reject) => { child.once('error', reject); child.once('exit', (code, signal) => resolve({ code, signal })); });
  clearInterval(monitor); clearTimeout(escalation); await log.close();
  await save('end.json', { ...result, minimumFreeMiB, stoppedForDisk, endedAt: new Date().toISOString(), logSha256: sha256(await readFile(join(root, 'run.log'))) });
  clean(); assert.equal(result.code, 0); assert.equal(result.signal, null); assert.equal(stoppedForDisk, false);
  console.log(JSON.stringify({ ...result, minimumFreeMiB, scope }));
} else {
  const author = join(root, 'author'), projectRoot = join(author, 'project'), outputRoot = join(author, 'release');
  const consumer = join(root, 'consumer'), handoff = join(consumer, 'handoff');
  const environment = finContainerEnvironment(['java']);
  environment.LEAN_BRIDGE_LEAN_PREFIX = '/app/.toolchains/elan/toolchains/leanprover--lean4---v4.32.2';
  environment.LEAN_BRIDGE_JAVA = '/app/.toolchains/jdk22/bin/java';
  environment.LEAN_BRIDGE_JAVAC = '/app/.toolchains/jdk22/bin/javac';
  await cp('tests/fixtures/onboarding/native-fin-containers', projectRoot, { recursive: true });
  await writeFile(join(projectRoot, 'FinContainers.lean'), await finContainerEdgeSource());
  await writeFile(join(projectRoot, 'lean-bridge.exports.json'), canonicalJson({ schemaVersion: 1, modules: ['FinContainers'], targets: Object.fromEntries([finContainerTargets.java]) }));
  console.log('Building the exact original-plus-edge FinContainers Maven package.');
  const built = await buildCanonicalProject({ projectRoot, outputRoot, targets: ['maven'], environment }).catch(error => { error.message += JSON.stringify(error.details); throw error; });
  const model = JSON.parse(await readFile(join(outputRoot, 'native/component/model.json')));
  assert.deepEqual(Object.fromEntries(model.exports.map(item => [item.name, item.refinements ?? null])), finContainerEdgeRefinements);
  await writeFile(join(root, 'model.json'), await readFile(join(outputRoot, 'native/component/model.json')), { flag: 'wx' });
  const receipt = await copyPackageSetHandoff(outputRoot, handoff);
  await verifyPackageSetReceipt({ receiptPath: join(handoff, 'package-set-receipt.json') });
  await save('build.json', { scope, revision, bindingIrSha256: built.bindingIrSha256, receipt,
    source: await readFile(join(projectRoot, 'FinContainers.lean'), 'utf8'),
    glibc: execFileSync('/usr/bin/getconf', ['GNU_LIBC_VERSION'], { encoding: 'utf8' }).trim(),
    packageFloor: environment.LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR });
  // Only this run's generated author/build staging is removed. Its prepared archives remain.
  await rm(author, { recursive: true });
  const packages = receipt.packages.filter(item => item.target === 'maven');
  console.log('Installing the source-free Java consumer.');
  const installed = await installCopiedConsumer({ profile: 'java', consumer, handoff, packages, environment,
    fixture: { source: () => finContainerEdgeConsumer('java'), success: 'fin-container-ok', expectedChecks: 14089,
      verifyInstalledPackage: verifyFinContainerEdgeArchiveClosure, installJvm: installFinContainerEdgeJvm } });
  await save('installed.json', installed);
  console.log('Running the unchanged production eight-column edge public and raw entry probes under local GDB.');
  let dispatch;
  try {
    dispatch = await repeatFinContainerEdges({ profile: 'java', consumer, handoff, packages, command: installed.command,
      measureDispatch: true, expectedModelSha256: sha256(await readFile(join(root, 'model.json'))),
      leanPrefix: environment.LEAN_BRIDGE_LEAN_PREFIX, toolchainEnvironment: environment,
      jvmEnvironment: installed.jvmEnvironment });
  } catch(error) {
    await save('baseline-failure.json', { name: error.name, message: error.message, stack: error.stack, details: error.details ?? null });
    throw error;
  }
  await save('baseline-dispatch.json', dispatch);
  clean(); console.log('Prepared actual JAR, public probe classes, native resources and original GDB configurations.');
}
