import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { readFile } from "node:fs/promises";
import { canonicalJson, sha256 } from "../src/capsule/node.mjs";
import { reviewedFinWasmIr } from "../tests/helpers/reviewed-fin-wasm-fixture.mjs";
import { reviewedFinWasmMismatches } from "../tests/helpers/reviewed-fin-wasm-mismatches.mjs";
import { reviewedFinWasmTypeScript } from "../tests/helpers/reviewed-fin-wasm-typescript.mjs";
const root = "/app/build/vo1438-wasm-refusals-d5a61df-r2";
const revision = "d5a61df812a10d4cf22077048e7fa582af35accd";
const queueBytes = await readFile(`${root}/queue.json`), endBytes = await readFile(`${root}/end.json`);
const queue = JSON.parse(queueBytes), end = JSON.parse(endBytes);
assert.equal(queue.revision, revision); assert.equal(end.code, 0); assert.equal(end.signal, null);
assert.equal(end.stoppedForDisk, false); assert.ok(end.minimumFreeMiB >= 1024);
assert.equal(queue.environment.LEAN_BRIDGE_REVIEWED_FIN_WASM_TEST, "1");
assert.equal(queue.environment.LEAN_BRIDGE_REVIEWED_FIN_WASM_BROWSER_TEST, "1");
assert.equal(queue.environment.LEAN_BRIDGE_TYPE_CORPUS_BROWSERS, "chromium,firefox,webkit");
assert.equal(queue.environment.LEAN_BRIDGE_REVIEWED_FIN_WASM_REPORT_DIR, root);
assert.equal(sha256(await readFile("build/run-vo1438-wasm-refusals-d5a61df-r2.mjs")), queue.runnerSha256);
for (const [path, digest] of Object.entries(queue.sources)) {
  assert.equal(sha256(execFileSync("git", ["show", `${revision}:${path}`], { maxBuffer: 16 * 1024 * 1024 })), digest, path);
  assert.equal(sha256(await readFile(`${queue.cwd}/${path}`)), digest, path);
}
for (const [path, digest] of Object.entries(queue.runtimeInputs)) assert.equal(sha256(await readFile(path)), digest, path);
assert.equal(Object.keys(queue.runtimeInputs).length, 4);
const compilerSha256 = sha256(await readFile(queue.environment.LEAN_BRIDGE_LEAN));
const tap = await readFile(`${root}/run.tap`);
assert.equal(sha256(tap), end.tapSha256);
for (const [label, count] of [["tests", 2], ["pass", 2], ["fail", 0], ["cancelled", 0], ["skipped", 0]])
  assert.match(tap.toString(), new RegExp(`^# ${label} ${count}$`, "mu"));
assert.doesNotMatch(tap.toString(), /^\s*not ok /mu);
const summaries = [];
for (const selection of ["scalar", "structural"]) {
  const bytes = await readFile(`${root}/reviewed-${selection}.json`), report = JSON.parse(bytes);
  assert.equal(report.path, "reviewed-ir"); assert.equal(report.selection, selection);
  assert.equal(report.privateAbi, selection === "scalar" ? 2 : 6); assert.equal(report.independentBuilds, 2);
  for (const flag of ["reproducible", "sourceRemovedBeforeInstallation", "compilerFreePath", "offlineInstall"]) assert.equal(report[flag], true);
  const expected = reviewedFinWasmIr(selection);
  assert.equal(report.reviewedSourceSha256, sha256(canonicalJson(expected)));
  assert.equal(report.sourceSha256, queue.sources["tests/fixtures/onboarding/reviewed-fin-wasm/ReviewedFin.lean"]);
  assert.equal(report.consumerSha256, queue.sources["tests/fixtures/reviewed-fin-wasm/javascript.mjs"]);
  assert.equal(report.compilerSha256, compilerSha256);
  assert.deepEqual(report.refinements, Object.fromEntries(expected.declarations.map(item => [item.source.declaration, item.source.extensions["lean-lang.org/refinements"]])));
  assert.deepEqual(report.mismatches, reviewedFinWasmMismatches(selection).map(({ label, ir, expectedField }) => ({ label,
    reviewedSourceSha256: sha256(canonicalJson(ir)), expectedField, fieldObserved: true, code: "reviewed-ir-source-mismatch", outputAbsent: true })));
  assert.equal(report.typescript.strict, true); assert.equal(report.typescript.skipLibCheck, false);
  assert.equal(report.typescript.sourceSha256, sha256(reviewedFinWasmTypeScript(selection)));
  assert.equal(report.receiptSha256, sha256(canonicalJson(report.receipt)));
  assert.equal(report.receipt.bindingIrSha256, report.bindingIrSha256);
  const result = { module: "reviewed-fin", selection, checks: selection === "scalar" ? 50 : 200, rejections: selection === "scalar" ? 126 : 270 };
  assert.deepEqual(report.executions.map(item => item.profile), ["node-javascript", "node-typescript"]);
  for (const item of report.executions) { assert.equal(item.hostVersion, queue.node); assert.deepEqual(item.result, result); }
  assert.deepEqual(report.browser.map(item => item.profile), ["browser-javascript", "browser-react", "browser-worker"]);
  let browserExecutions = 0;
  for (const { profile, browser } of report.browser) {
    assert.deepEqual(browser.requestedEngines, ["chromium", "firefox", "webkit"]);
    assert.equal(browser.externalNetworkBlocked, true); assert.equal(browser.installedSourcesRemoved, true);
    const variants = profile === "browser-react" ? ["production", "strict"] : ["production"];
    assert.deepEqual(browser.executions.map(item => `${item.engine}:${item.variant}`), browser.requestedEngines.flatMap(engine => variants.map(variant => `${engine}:${variant}`)));
    for (const execution of browser.executions) {
      browserExecutions++;
      assert.equal(execution.observation.profile, profile); assert.deepEqual(execution.observation.results, result);
      assert.equal(execution.observation.realm, profile === "browser-worker" ? "dedicated-worker" : "window");
      assert.ok(execution.observation.hostVersion.length > 0); assert.equal(execution.failedAssetRecovery, true);
      assert.ok(execution.assets.length >= 2);
      assert.deepEqual([...new Set(execution.assets.map(asset => asset.sha256))].sort(),
        [queue.runtimeInputs["/app/build/lean-link-spike/lazy/main.wasm"], report.receipt.componentArtifactSha256].sort());
      for (const asset of execution.assets) { assert.equal(asset.status, 200); assert.equal(asset.mime, "application/wasm"); assert.match(asset.sha256, /^[a-f0-9]{64}$/u); }
      if (profile === "browser-react") assert.equal(execution.pendingUnmount, true);
      if (profile === "browser-worker") assert.deepEqual(execution.lifecycle, { created: 2, live: 0, terminated: 2 });
    }
  }
  assert.equal(browserExecutions, 12);
  summaries.push({ selection, reportSha256: sha256(bytes), mismatches: report.mismatches.length, fieldsActuallyObserved: true,
    nodeExecutions: report.executions.length, browserExecutions, checksPerConsumer: result.checks, rejectionsPerConsumer: result.rejections });
}
console.log(JSON.stringify({ revision, queueSha256: sha256(queueBytes), endSha256: sha256(endBytes), tapSha256: sha256(tap),
  sourcePins: Object.keys(queue.sources).length, runtimePins: Object.keys(queue.runtimeInputs).length, minimumFreeMiB: end.minimumFreeMiB,
  scope: queue.scope, reports: summaries }, null, 2));
