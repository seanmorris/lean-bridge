let stdout = [], stderr = [];
const results = [], record = { scope, driverSha256: sha256(await readFile(new URL(import.meta.url))), results };
php.addEventListener('output', event => stdout.push([...event.detail]));
php.addEventListener('error', event => stderr.push([...event.detail]));
const persist = () => writeFile(join(root, 'smoke-results.json'), JSON.stringify(record, null, 2) + '\n');
const execute = async (label, source, expected) => {
  stdout = []; stderr = [];
  const status = await php.run(source);
  const output = stdout.flat().join('');
  const observation = { label, status, stdout, stderr, sourceSha256: sha256(source), source };
  results.push(observation);
  await persist();
  assert.equal(status, 0, JSON.stringify(observation));
  assert.deepEqual(JSON.parse(output), expected, label);
  const lines = stderr.flat();
  if (label === 'autoload') assert.deepEqual(lines, []);
  else {
    const first = /^LB_SUBTYPE_ENTRY_V1 begin public ([A-Za-z_][A-Za-z0-9_]*)$/.exec(lines[0]);
    assert.ok(first, 'Missing public begin marker: ' + label);
    assert.equal(lines.at(-1), 'LB_SUBTYPE_ENTRY_V1 end public ' + first[1]);
    const entries = lines.slice(1, -1).map(line => {
      const match = /^LB_SUBTYPE_ENTRY_V1 enter (constructor|adapter|source|validator) ([A-Za-z_][A-Za-z0-9_.:]*)$/.exec(line);
      assert.ok(match, 'Unexpected probe stderr: ' + line);
      assert.equal(selected.filter(item => item.kind === match[1] && item.label === match[2]).length, 1);
      return { kind: match[1], label: match[2] };
    });
    observation.entries = entries;
    observation.counts = Object.fromEntries(['constructor', 'validator', 'adapter', 'source'].map(kind => [kind, entries.filter(item => item.kind === kind).length]));
  }
  await persist();
  console.log(JSON.stringify({ label, result: JSON.parse(output), counts: observation.counts, stderr }));
};
