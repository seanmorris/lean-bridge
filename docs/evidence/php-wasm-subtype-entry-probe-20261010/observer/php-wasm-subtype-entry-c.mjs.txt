/**
 * Reversible, test-only entry markers in generated C for the PHP-Wasm Subtype probe.
 * Instrumented packages are separate observations, never unmodified release measurements.
 *
 * @file
 */
import assert from "node:assert/strict";
import { sha256 } from "../../src/capsule/node.mjs";

const kinds = ["public", "validator", "constructor", "adapter", "source"];

/**
 * Validate the explicit observer selection without inferring source entries from adapters.
 *
 * @param selected - Distinct C symbols with independently named entry kinds and labels.
 */
export const validateSubtypeEntrySelection = selected => {
	assert.ok(Array.isArray(selected) && selected.length > 0);
	for(const item of selected)
	{
		assert.deepEqual(Object.keys(item).sort(), ["kind", "label", "symbol"]);
		assert.ok(kinds.includes(item.kind));
		assert.match(item.symbol, /^[A-Za-z_][A-Za-z0-9_]*$/u);
		assert.match(item.label, /^[A-Za-z_][A-Za-z0-9_.:]*$/u);
	}
	assert.equal(new Set(selected.map(item => item.symbol)).size, selected.length);
	assert.equal(new Set(selected.map(item => `${item.kind}/${item.label}`)).size, selected.length);
};

/**
 * Hide comments, quoted literals and preprocessor directives while preserving source offsets.
 * This recognizes the generated C subset, not arbitrary macro-expanded C declarations.
 *
 * @param source - Complete generated C translation unit.
 */
const maskedC = source => source.replace(/\/\*[\s\S]*?\*\/|\/\/[^\n]*|"(?:\\[\s\S]|[^"\\])*"|'(?:\\[\s\S]|[^'\\])*'|^[\t ]*#(?:[^\n\\]|\\[\s\S])*/gmu, value => value.replace(/[^\n]/gu, " "));

/**
 * Mark selected definitions, retaining their original signatures and every original body byte.
 * A caller must require exactly one definition across all compiled translation units per selection.
 *
 * @param original - Complete original generated C.
 * @param selected - Full validated entry selection for this probe.
 */
export const instrumentSubtypeCEntries = (original, selected) => {
	assert.equal(typeof original, "string");
	validateSubtypeEntrySelection(selected);
	assert.doesNotMatch(original, /LB_SUBTYPE_ENTRY_V1|lb_probe_public_scope|lb_probe_end/u);
	const masked = maskedC(original), insertions = [];
	for(const item of selected)
	{
		const pattern = new RegExp(`^[A-Za-z_][A-Za-z0-9_* \\t]*\\b${item.symbol}\\([^{};]*\\)\\s*\\{`, "gmu");
		const matches = [...masked.matchAll(pattern)];
		assert.ok(matches.length <= 1, `duplicate selected definition: ${item.symbol}`);
		if(matches.length === 0) continue;
		const text = item.kind === "public"
			? `\n  const char *lb_probe_public_scope __attribute__((cleanup(lb_probe_end))) = ${JSON.stringify(item.label)};\n  fputs(${JSON.stringify(`LB_SUBTYPE_ENTRY_V1 begin public ${item.label}\n`)}, stderr);\n`
			: `\n  fputs(${JSON.stringify(`LB_SUBTYPE_ENTRY_V1 enter ${item.kind} ${item.label}\n`)}, stderr);\n`;
		insertions.push({ offset: matches[0].index + matches[0][0].length, text, ...item });
	}
	if(insertions.length > 0)
	{
		const helper = insertions.some(item => item.kind === "public")
			? 'static void lb_probe_end(const char **label) { fprintf(stderr, "LB_SUBTYPE_ENTRY_V1 end public %s\\n", *label); }\n' : "";
		insertions.push({ offset: 0, text: "#include <stdio.h>\n" + helper, kind: "preamble" });
	}
	insertions.sort((left, right) => right.offset - left.offset);
	let source = original;
	for(const { offset, text } of insertions) source = source.slice(0, offset) + text + source.slice(offset);
	const receipt = { originalSha256: sha256(original), probeSha256: sha256(source), insertions };
	assert.equal(restoreSubtypeCEntries(source, receipt), original);
	return { source, receipt };
};

/**
 * Restore an exact original unit from authenticated probe bytes and recorded insertion spans.
 * The caller separately authenticates the receipt and the original compiler output.
 *
 * @param source - Complete instrumented C unit.
 * @param receipt - Recorded original/probe digests and insertion spans.
 */
export const restoreSubtypeCEntries = (source, receipt) => {
	assert.equal(sha256(source), receipt.probeSha256);
	assert.ok(Array.isArray(receipt.insertions));
	let restored = source, cursor = -1;
	for(const { offset, text } of [...receipt.insertions].reverse())
	{
		assert.ok(Number.isSafeInteger(offset) && offset > cursor && offset <= restored.length);
		assert.ok(typeof text === "string" && text.length > 0);
		assert.equal(restored.slice(offset, offset + text.length), text);
		restored = restored.slice(0, offset) + restored.slice(offset + text.length);
		cursor = offset;
	}
	assert.equal(sha256(restored), receipt.originalSha256);
	return restored;
};
