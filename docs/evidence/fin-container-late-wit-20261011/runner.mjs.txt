import assert from 'node:assert/strict';
import { spawn, execFileSync } from 'node:child_process';
import { openSync, closeSync } from 'node:fs';
import { readFile, statfs, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
const prefix = process.argv[2];
assert.match(prefix, /^build\/vo1454-late-wit-installed-r[0-9]+$/);
const git = args => execFileSync('git', args, {encoding:'utf8'}).trim();
assert.equal(git(['diff','HEAD','--name-only']), '');
const free = async () => {const s=await statfs('/app');return Math.floor(s.bavail*s.bsize/1048576);};
let minimumMiB=await free();assert.ok(minimumMiB>=2048);
const environment={LEAN_BRIDGE_FIN_LATE_WIT_TEST:'1',LEAN_BRIDGE_FIN_LATE_WIT_OUTPUT:prefix+'-records',LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR:'2.36',LEAN_NUM_THREADS:'1'};
const args=['-c','3',process.execPath,'--test','--test-concurrency=1','tests/helpers/fin-container-late-wit-installed-tests.mjs'];
await writeFile(prefix+'.start.json',JSON.stringify({revision:git(['rev-parse','HEAD']),started:new Date().toISOString(),command:'/usr/bin/taskset',args,environment,minimumMiB},null,2)+'\n',{flag:'wx'});
const fd=openSync(prefix+'.tap','wx');
const child=spawn('/usr/bin/taskset',args,{cwd:'/app',env:{...process.env,...environment},stdio:['ignore',fd,fd],detached:true});
let diskStop=false;
const timer=setInterval(async()=>{const mib=await free();minimumMiB=Math.min(minimumMiB,mib);if(mib<768&&!diskStop){diskStop=true;process.kill(-child.pid,'SIGTERM');}process.stdout.write(JSON.stringify({time:new Date().toISOString(),pid:child.pid,freeMiB:mib,diskStop})+'\n');},10000);
child.on('error',error=>{process.stderr.write(String(error));process.exitCode=1;});
const end=await new Promise(resolve=>child.on('close',(code,signal)=>resolve({code,signal})));
clearInterval(timer);closeSync(fd);
const tap=await readFile(prefix+'.tap');
const result={...end,finished:new Date().toISOString(),minimumMiB,diskStop,tapSha256:createHash('sha256').update(tap).digest('hex')};
await writeFile(prefix+'.end.json',JSON.stringify(result,null,2)+'\n',{flag:'wx'});
process.stdout.write(JSON.stringify(result)+'\n');
process.exitCode=end.code===0&&!diskStop?0:1;
