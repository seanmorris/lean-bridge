import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { execFileSync, spawn } from "node:child_process";
import { closeSync, openSync, writeSync } from "node:fs";
import { readFile, statfs } from "node:fs/promises";

// Adapted from build/vo1441-python311-queue-72c5e27.mjs for the rust profile at 78a4d3d (VO #1441/#1442, board 1777).
const revision = "78a4d3da45754f3e425faafdadc9e959cee65bc5";
const cwd = "/app/build/worktrees/checked-records-vo1220";
const prefix = "/app/build/vo1441-rust-78a4d3d";
const rust = "/app/.toolchains/rust-1.90.0/bin";
const lean = "/app/.toolchains/elan/toolchains/leanprover--lean4---v4.32.2";
const git = args => execFileSync("git", args, { cwd, encoding: "utf8" }).trim();
const version = (command, args = ["--version"]) => execFileSync(command, args, { encoding: "utf8" }).split("\n")[0].trim();
assert.equal(git(["rev-parse", "HEAD"]), revision);
assert.equal(git(["status", "--porcelain", "--untracked-files=no"]), "");
const queue = openSync(`${prefix}.queue`, "wx");
const tap = openSync(`${prefix}.tap`, "wx");
const stamp = () => new Date().toISOString();
const log = line => { writeSync(queue, line + "\n"); process.stdout.write(line + "\n"); };
const freeMiB = async () => { const fs = await statfs(cwd); return Math.floor(fs.bavail * fs.bsize / 1024 / 1024); };
const base = Object.fromEntries(Object.entries(process.env).filter(([key]) => !key.startsWith("LEAN_BRIDGE_")));
const environment = { ...base, LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR: "2.36"
  , LEAN_BRIDGE_CARGO: `${rust}/cargo`, LEAN_BRIDGE_RUSTC: `${rust}/rustc`, LEAN_BRIDGE_LEAN_PREFIX: lean
  , LEAN_NUM_THREADS: "1", OMP_NUM_THREADS: "1", MAKEFLAGS: "-j1" };
const fixtures = [
  ["product", "PRODUCT", "native-fin-products", "^relocated source-free native packages check Fin inside products", "^independently reviewed native packages check product and Except bounds"]
  , ["product-array", "PRODUCT_ARRAY", "native-fin-product-arrays", "^relocated source-free native packages check Fin inside every element", "^independently reviewed native packages check array-of-product bounds"]
  , ["record", "RECORD", "native-fin-records", "^relocated source-free native packages check Fin inside record fields", "^independently reviewed native packages check record and variant field bounds"]
];
log(`revision=${revision} cwd=${cwd} runner=${prefix.replace(/78a4d3d$/u, "queue-78a4d3d")}.mjs`);
log(`node=${process.version} cargo=${version(`${rust}/cargo`)} rustc=${version(`${rust}/rustc`)} lean=${version(`${lean}/bin/lean`)} glibc=${version("/usr/bin/ldd")} cc=${version("/usr/bin/cc")}`);
log(`env LEAN_BRIDGE_NATIVE_TEST_GLIBC_FLOOR=2.36 LEAN_BRIDGE_CARGO=${rust}/cargo LEAN_BRIDGE_RUSTC=${rust}/rustc LEAN_BRIDGE_LEAN_PREFIX=${lean} LEAN_NUM_THREADS=1 OMP_NUM_THREADS=1 MAKEFLAGS=-j1; all inherited LEAN_BRIDGE_* removed; cpu=3 concurrency=1`);
log(`start=${stamp()} freeMiB=${await freeMiB()}`);
try {
  for(const [name, variable, testFile, ordinaryPattern, reviewedPattern] of fixtures) {
    for(const reviewed of [false, true]) {
      const free = await freeMiB();
      assert.ok(free >= 2048, `Only ${free}MiB free; refusing next producer`);
      assert.equal(git(["rev-parse", "HEAD"]), revision);
      assert.equal(git(["status", "--porcelain", "--untracked-files=no"]), "");
      const label = `${name}-${reviewed ? "reviewed" : "ordinary"}`;
      const selected = `LEAN_BRIDGE_${reviewed ? "REVIEWED_" : ""}FIN_${variable}`;
      const report = `/app/build/vo1441-rust-${label}-78a4d3d.json`;
      const pattern = reviewed ? reviewedPattern : ordinaryPattern;
      const env = { ...environment, [`${selected}_PROFILES`]: "rust", [`${selected}_REPORT`]: report };
      log(`start ${label} ${stamp()} freeMiB=${free} file=tests/${testFile}.test.mjs pattern=${pattern} ${selected}_PROFILES=rust ${selected}_REPORT=${report}`);
      writeSync(tap, `# step ${label}\n`);
      let output = "";
      const code = await new Promise((resolve, reject) => {
        const child = spawn("/usr/bin/taskset", ["-c", "3", process.execPath, "--test", "--test-concurrency=1", "--test-reporter=tap", `--test-name-pattern=${pattern}`, `tests/${testFile}.test.mjs`], { cwd, env, stdio: ["ignore", "pipe", "pipe"] });
        const record = chunk => { writeSync(tap, chunk); output += chunk.toString(); };
        child.stdout.on("data", record);
        child.stderr.on("data", record);
        child.once("error", reject);
        child.once("close", (code, signal) => resolve(code ?? (signal ? 128 : 1)));
      });
      writeSync(tap, `exit=${code}\n`);
      const count = key => Number(output.match(new RegExp(`^# ${key} (\\d+)$`, "mu"))?.[1]);
      log(`end ${label} ${stamp()} exit=${code} pass=${count("pass")} fail=${count("fail")} skipped=${count("skipped")} freeMiB=${await freeMiB()}`);
      assert.equal(code, 0, `${label} failed; queue stopped`);
      // Each selection must execute: exactly one passing test and nothing skipped.
      assert.deepEqual([count("pass"), count("fail"), count("skipped")], [1, 0, 0], `${label} did not execute exactly one test; queue stopped`);
      const bytes = await readFile(report), data = JSON.parse(bytes);
      assert.equal(data.reproducible, true);
      assert.deepEqual(data.reports.map(row => row.profile), ["rust"]);
      assert.equal(data.reports[0].sourceRemovedBeforeInstallation, true);
      log(`report ${label} sha256=${createHash("sha256").update(bytes).digest("hex")} checks=${data.reports[0].checks} dispatch=${JSON.stringify(data.reports[0].dispatch ?? null)}`);
    }
  }
  log(`all steps passed ${stamp()}`);
} catch(error) {
  log(`stopped ${stamp()} ${error.message}`);
  process.exitCode = 1;
} finally {
  closeSync(tap); closeSync(queue);
}
