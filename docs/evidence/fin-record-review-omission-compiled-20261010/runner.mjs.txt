import assert from "node:assert/strict";
import { spawn, execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { open, readFile, statfs, writeFile } from "node:fs/promises";
import { setTimeout as delay } from "node:timers/promises";

const cwd = "/app/build/worktrees/fin-container-edge-file-closure-vo1454";
const revision = "179d385755cda5908cdc63dd4cc5a39989acbbad";
const prefix = "/app/build/vo1442-record-omission-179d385";
const queuePid = 2677009, queueCommand = "/app/build/vo1454-foreign-installed-queue-3ca155f.mjs";
const git = args => execFileSync("git", args, { cwd, encoding: "utf8" }).trim();
const hash = bytes => createHash("sha256").update(bytes).digest("hex");
const freeMiB = async () => { const fs = await statfs("/app"); return Math.floor(fs.bavail * fs.bsize / 1048576); };
const queueLive = async () => {
  try {
    const command = await readFile("/proc/" + queuePid + "/cmdline", "utf8");
    assert.ok(command.split("\0").includes(queueCommand), "Refuse a reused queue PID");
    return true;
  } catch(error) { if(error.code === "ENOENT") return false; throw error; }
};
await writeFile(prefix + ".queued.json", JSON.stringify({
  revision, cwd, pid: process.pid, queuedAt: new Date().toISOString(), waitForPid: queuePid, waitForCommand: queueCommand
}, null, 2) + "\n", { flag: "wx" });
console.log("Waiting for the original installed queue PID " + queuePid);
while(await queueLive()) await delay(5000);
for(const label of ["c-cpp", "python311", "python312", "rust", "ruby", "dotnet", "java-kotlin", "php-native", "wit-wasi"]) {
  const base = "/app/build/vo1454-foreign-installed-" + label + "-3ca155f";
  const end = JSON.parse(await readFile(base + ".end.json", "utf8"));
  const verified = JSON.parse(await readFile(base + ".verified.json", "utf8"));
  assert.deepEqual([end.code, end.signal, end.stoppedForDisk], [0, null, false]);
  assert.equal(verified.revision, "3ca155faffbe5ee9ea2a491bf28728579c10f2d5");
  assert.equal(hash(await readFile(verified.report)), verified.reportSha256);
}
assert.equal(git(["rev-parse", "HEAD"]), revision);
assert.equal(git(["status", "--porcelain", "--untracked-files=no"]), "");
const initialFreeMiB = await freeMiB();
assert.ok(initialFreeMiB >= 2048, "Refuse a new heavy producer below 2 GiB free");
const environment = {
  LEAN_BRIDGE_REVIEWED_FIN_RECORD_PROFILES: "c",
  LEAN_BRIDGE_LEAN_PREFIX: "/app/.toolchains/elan/toolchains/leanprover--lean4---v4.32.2",
  LEAN_NUM_THREADS: "1", OMP_NUM_THREADS: "1", MAKEFLAGS: "-j1"
};
const args = ["-c", "3", process.execPath, "--test", "--test-concurrency=1", "--test-reporter=tap",
  "--test-name-pattern=^changed record and variant reviews are refused against fresh Lean before any output$",
  "tests/native-fin-records.test.mjs"];
const log = await open(prefix + ".tap", "wx");
const child = spawn("/usr/bin/taskset", args, { cwd, env: { ...process.env, ...environment }, detached: true, stdio: ["ignore", log.fd, log.fd] });
await writeFile(prefix + ".start.json", JSON.stringify({
  revision, tree: git(["rev-parse", "HEAD^{tree}"]), cwd, pid: child.pid,
  command: "/usr/bin/taskset", args, environment, initialFreeMiB, startedAt: new Date().toISOString()
}, null, 2) + "\n", { flag: "wx" });
console.log("Started fresh Lean omission gate PID " + child.pid);
let minimumFreeMiB = initialFreeMiB, stoppedForDisk = false, finished = false;
const monitor = (async () => {
  while(!finished) {
    minimumFreeMiB = Math.min(minimumFreeMiB, await freeMiB());
    if(minimumFreeMiB < 768 && !stoppedForDisk) {
      stoppedForDisk = true; process.kill(-child.pid, "SIGTERM");
    }
    await delay(1000);
  }
})();
const outcome = await new Promise((resolve, reject) => { child.once("error", reject); child.once("exit", (code, signal) => resolve({ code, signal })); });
finished = true; await monitor; await log.close();
const bytes = await readFile(prefix + ".tap"), tap = bytes.toString("utf8");
const result = { ...outcome, stoppedForDisk, minimumFreeMiB, endedAt: new Date().toISOString(), tapSha256: hash(bytes) };
await writeFile(prefix + ".end.json", JSON.stringify(result, null, 2) + "\n", { flag: "wx" });
console.log(JSON.stringify(result));
assert.deepEqual([outcome.code, outcome.signal, stoppedForDisk], [0, null, false]);
for(const [name, count] of [["tests", 8], ["pass", 8], ["fail", 0], ["skipped", 0]])
  assert.equal(Number(tap.match(new RegExp("^# " + name + " (\\d+)$", "m"))?.[1]), count);
for(const name of ["tightened field", "bound moved to the other field", "loosened nested record's own bound", "tightened case field", "loosened Fin 0 case", "omitted record bounds", "omitted variant bounds"])
  assert.ok(tap.split("\n").some(line => new RegExp("^    ok \\d+ - " + name + "$").test(line)), name);
assert.equal(git(["rev-parse", "HEAD"]), revision);
assert.equal(git(["status", "--porcelain", "--untracked-files=no"]), "");
await writeFile(prefix + ".verified.json", JSON.stringify({ revision, tapSha256: hash(bytes), cases: 7, verifiedAt: new Date().toISOString() }, null, 2) + "\n", { flag: "wx" });
console.log("All seven fresh-Lean record and variant mutations were refused before output.");
